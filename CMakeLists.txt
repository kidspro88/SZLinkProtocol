cmake_minimum_required(VERSION 3.13)

project(szlink)

find_package(Python COMPONENTS Interpreter REQUIRED)

# We automatically install the pip dependencies locally below.
# Therefore, we check whether pip is available here.
execute_process(
    COMMAND ${Python_EXECUTABLE} -m pip -V
    RESULT_VARIABLE EXIT_CODE
    OUTPUT_QUIET
)
if (NOT ${EXIT_CODE} EQUAL 0)
    message(FATAL_ERROR "Python pip not found, pip is required")
endif()

if (NOT SZLINK_DIALECT)
    set(SZLINK_DIALECT common)
endif()
message(STATUS "SZLink dialect: ${SZLINK_DIALECT}")

if (NOT SZLINK_VERSION)
    set(SZLINK_VERSION 2.0)
endif()
message(STATUS "SZLink version: ${SZLINK_VERSION}")

# The headers are generated by scripts/szlinkgen.py: it runs pymavlink's mavgen
# and renames the generated API to szlink_* / SZLINK_* / namespace szlink, and
# adds the SZ-Link frame layer (szlink_frame.h, szlink_ecc.h, szlink_sec.h and
# Monocypher: monocypher.h + monocypher.c, installed next to them; compile
# monocypher.c into the program that uses szlink_sec.h). The frame layer needs
# SZLINK_VERSION 2.0; with 1.0 only the message headers are generated.
# With SZLINK_ENABLE_CXX it generates the C++11 headers (*.hpp); the C++11
# generator also emits the C headers (*.h), so both examples in examples/ build
# against the same install.
option(SZLINK_ENABLE_CXX "Also generate and install the C++11 (*.hpp) headers" ON)

if (SZLINK_ENABLE_CXX AND NOT SZLINK_VERSION VERSION_EQUAL 2.0)
    message(FATAL_ERROR
        "SZLINK_ENABLE_CXX=ON needs SZLINK_VERSION 2.0 (the C++11 generator only supports "
        "wire protocol 2.0), but SZLINK_VERSION is ${SZLINK_VERSION}. "
        "Configure with -DSZLINK_ENABLE_CXX=OFF for ${SZLINK_VERSION} C headers, "
        "or with -DSZLINK_VERSION=2.0.")
endif()

if (SZLINK_ENABLE_CXX)
    set(SZLINK_LANG C++11)
else()
    set(SZLINK_LANG C)
endif()

set(SZLINK_INCLUDE_DIR ${CMAKE_CURRENT_BINARY_DIR}/include/szlink)
set(EXAMPLE_HEADER ${SZLINK_INCLUDE_DIR}/${SZLINK_DIALECT}/szlink.h)

add_custom_command(OUTPUT ${EXAMPLE_HEADER}
    COMMAND ${Python_EXECUTABLE}
        -m pip install -r pymavlink/requirements.txt --upgrade -t ${CMAKE_CURRENT_BINARY_DIR}/pip-dependencies/
    COMMAND ${CMAKE_COMMAND} -E env "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/pip-dependencies/" ${Python_EXECUTABLE}
        scripts/szlinkgen.py
        --lang=${SZLINK_LANG}
        --wire-protocol=${SZLINK_VERSION}
        --output ${SZLINK_INCLUDE_DIR}
        message_definitions/v1.0/${SZLINK_DIALECT}.xml
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    DEPENDS message_definitions/v1.0/${SZLINK_DIALECT}.xml scripts/szlinkgen.py
        szlink/c/szlink_frame.h szlink/c/szlink_ecc.h szlink/c/szlink_sec.h
        szlink/c/third_party/monocypher/monocypher.h szlink/c/third_party/monocypher/monocypher.c
    COMMENT "Generating SZLink ${SZLINK_LANG} headers")

# Unfortunately, the dependencies don't work for INTERFACE libraries.
# The only way I could make it work is to add ALL which means it
# will do the file generation every time even when nothing has changed.
add_custom_target(generate_c_headers
    ALL
    DEPENDS ${EXAMPLE_HEADER})

add_library(szlink INTERFACE)

add_dependencies(szlink generate_c_headers)

include(GNUInstallDirs)

target_include_directories(szlink
    INTERFACE
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

install(TARGETS szlink
    EXPORT SZLinkTargets
    PUBLIC_HEADER DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
)

install(DIRECTORY "${SZLINK_INCLUDE_DIR}"
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
    FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp" PATTERN "monocypher.c"
)

install(EXPORT SZLinkTargets
    FILE SZLinkTargets.cmake
    NAMESPACE SZLink::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SZLink
)

# For the build tree
configure_file(SZLinkConfig.cmake.in
    "${PROJECT_BINARY_DIR}/SZLinkConfig.cmake" @ONLY)
# For the install tree
configure_file(SZLinkConfig.cmake.in
    "${PROJECT_BINARY_DIR}${CMAKE_FILES_DIRECTORY}/SZLinkConfig.cmake" @ONLY)

install(FILES
    "${PROJECT_BINARY_DIR}/${CMAKE_FILES_DIRECTORY}/SZLinkConfig.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SZLink
)
